function createCard(item) {
  const card = document.createElement("article");
  card.className = "card media-card";
  const link = document.createElement("a");
  link.className = "card-link";
  link.href = item.url;
  link.target = "_blank";
  link.rel = "noopener noreferrer";
  link.setAttribute("aria-label", `${item.title}: ${item.detail} (opens in a new tab)`);

  const thumb = document.createElement("div");
  thumb.className = "thumb";
  thumb.style.setProperty("--tone-a", item.colors[0]);
  thumb.style.setProperty("--tone-b", item.colors[1]);
  const icon = document.createElement("span");
  icon.className = "thumb-icon";
  icon.setAttribute("aria-hidden", "true");
  icon.textContent = item.icon;
  thumb.append(icon);

  const info = document.createElement("div");
  info.className = "card-info";
  const title = document.createElement("h3");
  title.textContent = item.title;
  const detail = document.createElement("p");
  detail.textContent = item.detail;
  const tag = document.createElement("span");
  tag.className = "pill";
  tag.textContent = item.tag;
  info.append(title, detail, tag);
  link.append(thumb, info);
  card.append(link);
  return card;
}

Object.entries(catalogs).forEach(([elementId, items]) => {

  items.forEach(item => grid.append(createCard(item)));

function showTab(id) {
  if (!document.getElementById(id)) return;
  activeTab = id;
  searchInput.value = "";
  activeQuery = "";
  sections.forEach(section => section.classList.toggle("hidden", section.id !== id));
  navButtons.forEach(button => {
    const selected = button.dataset.tab === id;
    button.classList.toggle("active", selected);
    if (selected) button.setAttribute("aria-current", "page");
    else button.removeAttribute("aria-current");
  });
  applySearch();
  window.scrollTo({ top: 0, behavior: "smooth" });
}

navButtons.forEach(button => button.addEventListener("click", () => showTab(button.dataset.tab)));

document.getElementById("searchForm").addEventListener("submit", event => {
  event.preventDefault();
  activeQuery = searchInput.value.trim().toLocaleLowerCase();
  applySearch();
});
searchInput.addEventListener("input", () => {
  activeQuery = searchInput.value.trim().toLocaleLowerCase();
  applySearch();
});

function applySearch() {
  if (!activeQuery) {
    sections.forEach(section => {
      section.classList.toggle("hidden", section.id !== activeTab);
      section.querySelectorAll(".media-card").forEach(card => card.classList.remove("hidden"));
      section.querySelector(".game-card")?.classList.remove("hidden");
    });
    searchStatus.textContent = "";
    return;
  }

  let totalMatches = 0;
  let matchingSections = 0;
  sections.forEach(section => {
    let sectionMatches = 0;
    section.querySelectorAll(".media-card").forEach(card => {
      const match = card.textContent.toLocaleLowerCase().includes(activeQuery);
      card.classList.toggle("hidden", !match);
      if (match) sectionMatches++;
    });
    const quiz = section.querySelector(".game-card");
    if (quiz) {
      const match = quiz.textContent.toLocaleLowerCase().includes(activeQuery);
      quiz.classList.toggle("hidden", !match);
      if (match) sectionMatches++;
    }
    section.classList.toggle("hidden", sectionMatches === 0);
    totalMatches += sectionMatches;
    if (sectionMatches) matchingSections++;
  });
  searchStatus.textContent = totalMatches
    ? `${totalMatches} matching result${totalMatches === 1 ? "" : "s"} across ${matchingSections} section${matchingSections === 1 ? "" : "s"}.`
    : "No matching results. Try another search.";
}

const accountForm = document.getElementById("accountForm");
const accountMessage = document.getElementById("accountMessage");
const clearProfile = document.getElementById("clearProfile");
const profileKey = "omnihub_profile";

function showSavedProfile() {
  try {
    const profile = JSON.parse(localStorage.getItem(profileKey) || "null");
    if (!profile || typeof profile.name !== "string" || typeof profile.email !== "string") return;
    document.getElementById("username").value = profile.name;
    document.getElementById("email").value = profile.email;
    accountMessage.textContent = `Profile saved on this device for ${profile.name}.`;
    clearProfile.classList.remove("hidden");
  } catch {
    accountMessage.textContent = "Saved profile could not be read in this browser.";
  }
}

accountForm.addEventListener("submit", event => {
  event.preventDefault();
  if (!accountForm.reportValidity()) return;
  const profile = {
    name: document.getElementById("username").value.trim(),
    email: document.getElementById("email").value.trim()
  };
  if (!profile.name || !profile.email) {
    accountMessage.textContent = "Enter your name and a valid email address.";
    return;
  }
  try {
    localStorage.setItem(profileKey, JSON.stringify(profile));
    accountMessage.textContent = "Profile saved on this device. This is not an online account.";
    clearProfile.classList.remove("hidden");
  } catch {
    accountMessage.textContent = "Could not save the profile. Browser storage may be unavailable.";
  }
});

clearProfile.addEventListener("click", () => {
  try {
    localStorage.removeItem(profileKey);
    accountForm.reset();
    accountMessage.textContent = "Saved profile removed from this device.";
    clearProfile.classList.add("hidden");
  } catch {
    accountMessage.textContent = "Could not remove the saved profile.";
  }
});
showSavedProfile();

const quizQuestions = [
  { question: "Which planet is known as the Red Planet?", options: ["Venus", "Mars", "Jupiter"], answer: "Mars" },
  { question: "How many sides does a hexagon have?", options: ["Five", "Six", "Eight"], answer: "Six" },
  { question: "Which is the largest ocean on Earth?", options: ["Atlantic", "Indian", "Pacific"], answer: "Pacific" }
];
let quizIndex = 0;
function renderQuestion() {
  const question = quizQuestions[quizIndex];
  document.getElementById("quizQuestion").textContent = question.question;
  document.getElementById("quizFeedback").textContent = "";
  const options = document.getElementById("quizOptions");
  options.replaceChildren();
  question.options.forEach(option => {
    const button = document.createElement("button");
    button.type = "button";
    button.textContent = option;
    button.addEventListener("click", () => {
      const feedback = document.getElementById("quizFeedback");
      if (option === question.answer) {
        feedback.textContent = "Correct! Loading the next question…";
        window.setTimeout(() => {
          quizIndex = (quizIndex + 1) % quizQuestions.length;
          renderQuestion();
        }, 900);
      } else {
        feedback.textContent = "Not quite — try another answer.";
      }
    });
    options.append(button);
  });
}
renderQuestion();
